module

public import Mathlib.GroupTheory.Coxeter.Basic

/-!
# The Artin group and its Coxeter projection

For any Coxeter matrix, with no restriction on its indexing type, the Artin
group is the group on one generator per index and the alternating braid
relations. An entry zero denotes infinity and contributes no relation.
The standard parabolics below are subgroups generated by the selected standard
generators; no identification with a separately presented group is used.
-/

@[expose] public section

namespace ArtinNormalisers.Artin

variable {I : Type*} (M : CoxeterMatrix I)

/-- A positive word interpreted in the free group on the standard generators. -/
def freeWord (w : List I) : FreeGroup I := (w.map FreeGroup.of).prod

/-- The alternating braid relator of length `M i j`. The convention for
alternating words is mathlib's: the first word ends at `j`. -/
def relation (i j : I) : FreeGroup I :=
  freeWord (CoxeterSystem.alternatingWord i j (M i j)) *
    (freeWord (CoxeterSystem.alternatingWord j i (M i j)))⁻¹

/-- Only finite matrix entries impose a braid relation. -/
def relations : Set (FreeGroup I) :=
  {r | ∃ i j, M i j ≠ 0 ∧ r = relation M i j}

/-- The standard Artin presentation associated to the Coxeter matrix `M`. -/
abbrev Group : Type _ := PresentedGroup (relations M)

/-- The canonical Artin generator indexed by `i`. -/
def generator (i : I) : Group M := PresentedGroup.of i

theorem closure_range_generator :
    Subgroup.closure (Set.range (generator M)) = ⊤ :=
  PresentedGroup.closure_range_of _

/-- Products of the canonical Artin generators along a positive word. -/
def wordProd (w : List I) : Group M := (w.map (generator M)).prod

@[simp]
theorem mk_freeWord (w : List I) :
    PresentedGroup.mk (relations M) (freeWord w) = wordProd M w := by
  simp [freeWord, wordProd, map_list_prod, List.map_map, Function.comp_def, generator,
    PresentedGroup.of]

/-- The defining finite alternating braid relations hold in the actual Artin group. -/
theorem braid_relation (i j : I) (h : M i j ≠ 0) :
    wordProd M (CoxeterSystem.alternatingWord i j (M i j)) =
      wordProd M (CoxeterSystem.alternatingWord j i (M i j)) := by
  have hr : relation M i j ∈ relations M := ⟨i, j, h, rfl⟩
  have he := PresentedGroup.one_of_mem hr
  simpa only [relation, map_mul, map_inv, mk_freeWord, mul_inv_eq_one] using he

private theorem lift_freeWord (w : List I) :
    FreeGroup.lift M.simple (freeWord w) = M.toCoxeterSystem.wordProd w := by
  simp [freeWord, CoxeterSystem.wordProd, map_list_prod, List.map_map,
    Function.comp_def]

private theorem projection_relations (r : FreeGroup I) (hr : r ∈ relations M) :
    FreeGroup.lift M.simple r = 1 := by
  obtain ⟨i, j, _, rfl⟩ := hr
  rw [relation, map_mul, map_inv, lift_freeWord, lift_freeWord, mul_inv_eq_one]
  simpa only [CoxeterSystem.braidWord, M.symmetric i j] using
    M.toCoxeterSystem.wordProd_braidWord_eq i j

/-- The natural homomorphism from the Artin presentation to its Coxeter group. -/
def projection : Group M →* M.Group :=
  PresentedGroup.toGroup (projection_relations M)

@[simp]
theorem projection_generator (i : I) : projection M (generator M i) = M.simple i :=
  PresentedGroup.toGroup.of _

@[simp]
theorem projection_wordProd (w : List I) :
    projection M (wordProd M w) = M.toCoxeterSystem.wordProd w := by
  simp [wordProd, CoxeterSystem.wordProd, map_list_prod, List.map_map,
    Function.comp_def]

/-- Surjectivity follows from the Coxeter simple generators, each of which has
its canonical Artin generator as a preimage. -/
theorem projection_surjective : Function.Surjective (projection M) := by
  apply MonoidHom.range_eq_top.mp
  apply top_unique
  rw [← M.subgroupClosure_range_simple]
  apply Subgroup.closure_le.mpr
  rintro _ ⟨i, rfl⟩
  exact ⟨generator M i, projection_generator M i⟩

/-- The standard Artin parabolic generated by the generators with indices in `J`. -/
def parabolic (J : Set I) : Subgroup (Group M) :=
  Subgroup.closure (generator M '' J)

/-- The standard Coxeter parabolic generated by the simple reflections in `J`. -/
def coxeterParabolic (J : Set I) : Subgroup M.Group :=
  Subgroup.closure (M.simple '' J)

/-- The image of a standard Artin parabolic is the corresponding Coxeter parabolic. -/
theorem map_parabolic (J : Set I) :
    (parabolic M J).map (projection M) = coxeterParabolic M J := by
  rw [parabolic, MonoidHom.map_closure, coxeterParabolic, Set.image_image]
  congr 1
  ext w
  simp only [Set.mem_image, Function.comp_apply, projection_generator]

/-- The pure Artin subgroup, defined as the kernel of the natural projection. -/
def pure : Subgroup (Group M) := (projection M).ker

end ArtinNormalisers.Artin
